// prompt: texto, ou lista de "parts" do Gemini (texto + imagens em base64).
async function ask(prompt, search = false) {
  const parts = typeof prompt === "string" ? [{ text: prompt }] : prompt;
  const body = JSON.stringify({
    contents: [{ parts }],
    ...(search ? { tools: [{ google_search: {} }] } : { generationConfig: { responseMimeType: "application/json" } }),
  });
  let r;
  for (let t = 0; t < 3; t++) { // repete se o Gemini estiver sobrecarregado
    r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent`, {
      method: "POST", headers: { "x-goog-api-key": GK, "content-type": "application/json" }, body,
    });
    if (r.ok || ![429, 500, 503].includes(r.status)) break;
    await new Promise((ok) => setTimeout(ok, 5000 * (t + 1)));
  }
  if (!r.ok) throw new Error(`Gemini ${r.status}: ${(await r.text()).slice(0, 300)}`);
  const cand = (await r.json()).candidates?.[0];
  const text = (cand?.content?.parts ?? []).filter((p) => p.text && !p.thought).map((p) => p.text).join("\n");
  const a = text.indexOf("{"), b = text.lastIndexOf("}");
  if (a < 0 || b < a) throw new Error("Resposta sem JSON");
  return JSON.parse(text.slice(a, b + 1));
}

async function findFlyer(m) {
  const j = await ask(
    `Hoje é ${today}. Procura o folheto de promoções do supermercado ${m.name} (Portugal) que comece DEPOIS de ${m.from} ou que já esteja em vigor. Usa sites de antevisão (blog200porcento.com, ocacapromocoes.pt, poupapilim.com) ou o site oficial. Extrai produtos alimentares com preço escrito nessas páginas (mínimo 8; preço por kg ou por unidade/embalagem). NUNCA estimes ou inventes preços; se não encontrares, devolve found=false. Responde APENAS com JSON: {"found":true,"from":"AAAA-MM-DD","to":"AAAA-MM-DD","flyer":"texto curto","sources":[{"title":"","url":""}],"promos":[{"id":"slug-unico","name":"","price":0.0,"per":"kg|un","label":"opcional ex.: 1,29 €/emb. 300 g","note":"opcional"}]}`,
    true
  );
  const ids = new Set((j.promos ?? []).map((p) => p.id));
  const ok = j.found && j.from > m.from && j.to >= today && j.promos?.length >= 8 && ids.size === j.promos.length &&
    j.sources?.length && j.sources.every((s) => /^https?:\/\//.test(s.url)) &&
    j.promos.every((p) => p.id && p.name && p.price > 0 && p.price < 100 && ["kg", "un"].includes(p.per));
  return ok ? j : null;
}

async function makeRecipes(m) {
  const list = m.promos.map((p) => `${p.id}: ${p.name} ${p.price}€/${p.per}`).join("; ");
  const j = await ask(
    `Cria 2 receitas portuguesas (PT-PT) para o supermercado ${m.name}, usando PRODUTOS EM PROMOÇÃO desta lista (id: nome preço): ${list}. Cada receita usa pelo menos 2 produtos da lista, indicados com "promo" (id exato) e "amount" (kg se per=kg, unidades se per=un). Os restantes ingredientes são de despensa (sem "promo"). Não repitas: ${m.history.join(", ") || "nenhuma"}. Quantidades realistas, 3 a 5 passos com tempos e temperaturas, segurança alimentar correta. Responde APENAS com JSON: {"recipes":[{"name":"","description":"","servings":"4 pessoas","time":"45 min","imageQuery":"palavras-chave em inglês para procurar uma foto do prato pronto, ex.: fried horse mackerel with boiled potatoes","ingredients":[{"item":"","qty":"","promo":"id","amount":0}],"steps":[""]}]}`
  );
  if (!(j.recipes?.length >= 2)) throw new Error("menos de 2 receitas");
  const byId = Object.fromEntries(m.promos.map((p) => [p.id, p]));
  return j.recipes.slice(0, 2).map((r, i) => {
    const ingredients = r.ingredients.map((x) => {
      if (!x.promo) return { item: x.item, qty: x.qty };
      const p = byId[x.promo];
      if (!p || !(x.amount > 0 && x.amount <= 5)) throw new Error(`promo inválida: ${x.promo}`);
      return { item: x.item || p.name, qty: x.qty, price: p.price, per: p.per, amount: x.amount, ...(p.label && { label: p.label }), note: p.note ?? "folheto" };
    });
    if (ingredients.filter((x) => x.price).length < 2 || !(r.steps?.length >= 3)) throw new Error("receita fraca");
    return { slot: i + 1, name: r.name, description: r.description, servings: r.servings, time: r.time, ingredients, steps: r.steps, imageQuery: String(r.imageQuery ?? r.name) };
  });
}

// Foto real (Pexels, licença livre) escolhida SEM IA: só aceita uma foto cuja descrição
// (alt) contém a maior parte das palavras-chave do prato. Sem correspondência → sem foto.
const STOP = new Set(["with", "and", "the", "served", "fresh", "dish"]);
const toks = (t) => (t.toLowerCase().match(/[a-z]+/g) ?? []).filter((w) => w.length > 2 && !STOP.has(w)).map((w) => w.replace(/(es|s)$/, ""));
async function makeImage(m, r) {
  if (!PK) return undefined;
  try {
    const raw = (r.imageQuery ?? "").split(/\s+/).filter(Boolean), words = toks(r.imageQuery ?? "");
    if (words.length < 2) throw new Error("sem palavras-chave");
    const need = Math.max(2, Math.ceil(words.length * 0.6));
    let best = null, top = 0;
    for (let n = raw.length; n >= 2 && !best; n--) { // pesquisa completa, depois mais curta
      const res = await fetch(`https://api.pexels.com/v1/search?query=${encodeURIComponent(raw.slice(0, n).join(" "))}&per_page=15&orientation=landscape`, { headers: { Authorization: PK } });
      if (!res.ok) throw new Error(`Pexels ${res.status}`);
      for (const p of (await res.json()).photos ?? []) {
        const alt = new Set(toks(p.alt ?? ""));
        const score = words.filter((w) => alt.has(w)).length;
        if (score >= need && score > top) { best = p; top = score; }
      }
    }
    if (!best) throw new Error("nenhuma foto com descrição correspondente");
    console.log("  foto escolhida:", best.alt);
    const img = await fetch(best.src.large);
    if (!img.ok) throw new Error(`download ${img.status}`);
    const name = `foto-${slug(m.name)}-${slug(r.name)}.webp`;
    const sharp = (await import("sharp")).default;
    await sharp(Buffer.from(await img.arrayBuffer())).resize(900, 900, { fit: "cover" }).webp({ quality: 78 }).toFile(path.join(IMG, name));
    return { image: `/meals/${name}`, credit: { name: best.photographer, url: best.url } };
  } catch (e) {
    console.warn(`  sem foto para "${r.name}":`, e.message);
    return undefined; // a app mostra um fundo neutro, nunca uma foto errada
  }
}

/** Data de hoje em Lisboa (AAAA-MM-DD), não em UTC. */
export function lisbonToday(now = new Date()): string {
  return now.toLocaleDateString("en-CA", { timeZone: "Europe/Lisbon" });
}

function cost(i: Ing): number | null {
  return i.price != null && i.amount != null ? i.price * i.amount : null;
}

function priceText(i: Ing): string {
  const c = cost(i);
  if (c === null) return "despensa";
  const unit = i.label ?? `${eur(i.price!)}/${i.per}`;
  return `${unit} → ${eur(c)}`;
}

export function getTodaySuggestions(today = lisbonToday()) {
  const meals = MARKETS.flatMap((m, mi) => {
    const status = today < m.from ? "upcoming" : today > m.to ? "expired" : "active";
    return m.recipes.map((r) => {
      const total = r.ingredients.reduce((s, i) => s + (cost(i) ?? 0), 0);
      return {
        id: mi * 10 + r.slot, // estável: a lista de compras não fica órfã
        supermarket: m.name,
        slot: r.slot,
        name: r.name,
        description: r.description,
        servings: r.servings,
        time: r.time,
        ingredients: r.ingredients.map((i) => ({
          item: i.item,
          qty: i.qty,
          price: priceText(i),
          note: i.note,
        })),
        steps: r.steps,
        image: r.image ?? null,
        imageCredit: r.imageCredit ?? null,
        totalCost: `≈ ${eur(total)}`,
        source: m.flyer,
        validFrom: m.from,
        validTo: m.to,
        status,
        sources: m.sources,
      };
    });
  });
  return { day: today, meals };
}

function parseCost(price: string): number | null {
  const afterArrow = price.split("→").pop() ?? price;
  const m = afterArrow.match(/([\d]+[.,][\d]{2})\s*€/);
  if (!m) return null;
  return parseFloat(m[1].replace(",", "."));
}

          <button
            onClick={() => setListOpen(true)}
            className="flex shrink-0 items-center gap-2 rounded-full bg-[#1c1a17] px-4 py-2.5 text-[12px] font-medium text-white active:scale-95"
          >
            <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8">
              <path d="M6 6h15l-1.5 9h-12z" />
              <path d="M6 6L5 3H2" />
              <circle cx="9" cy="20" r="1.4" />
              <circle cx="17" cy="20" r="1.4" />
            </svg>
            Lista
            {basket.size > 0 && (
              <span className="flex h-5 min-w-5 items-center justify-center rounded-full bg-[#c14e1d] px-1.5 text-[10px] font-semibold leading-none text-white">
                {basket.size}
              </span>
            )}
          </button>
        </div>
      </header>

      <div className="mx-auto max-w-2xl">
        {/* ─── greeting ─── */}
        <section className="px-4 pb-2 pt-7">
          <h1 className="font-display text-[28px] font-semibold leading-tight tracking-tight">
            {greeting()} 👋
          </h1>
          <p className="mt-1 text-[15px] font-light text-[#1c1a17]/60">
            {data?.day ? formatDay(data.day) : "Hoje"} · preços de folheto verificados
          </p>
        </section>

        {/* ─── supermarket segmented control ─── */}
        <section className="sticky top-14 z-20 -mx-0 bg-[#f6f5f2]/95 px-4 py-3 backdrop-blur-md">
          <div className="no-scrollbar flex gap-1.5 overflow-x-auto rounded-2xl bg-[#ebe8e1] p-1.5">
            {SUPERMARKETS.map((s) => {
              const on = active === s.name;
              return (
                <button
                  key={s.name}
                  onClick={() => setActive(s.name)}
                  className={`shrink-0 rounded-xl px-4 py-2.5 text-[13px] font-medium transition-all active:scale-95 ${
                    on ? "bg-white text-[#1c1a17] shadow-sm" : "text-[#1c1a17]/50"
                  }`}
                >
                  <span
                    className="mr-1.5 inline-block h-2 w-2 rounded-full align-middle"
                    style={{ background: s.accent }}
                  />
                  {s.name}
                </button>
              );
            })}
          </div>
          <p className="mt-2.5 px-1 text-[12px] font-light text-[#1c1a17]/45">{market.note}</p>
        </section>

        {/* ─── meal cards ─── */}
        <main className="mt-2 space-y-5 px-4">
          {isLoading && (
            <p className="py-24 text-center font-mono-x text-sm text-[#1c1a17]/45">
              A preparar o menu…
            </p>
          )}
          {isError && (
            <p className="py-24 text-center text-sm text-[#c8102e]">
              Não foi possível carregar. Tenta novamente dentro de momentos.
            </p>
          )}

          {meals.map((m, i) => {
            const inBasket = basket.has(m.id);
            return (
              <article
                key={m.id}
                className="fade-swap card-shadow overflow-hidden rounded-3xl bg-white"
                style={{ animationDelay: `${i * 90}ms` }}
              >
                <button onClick={() => setOpenMeal(m)} className="block w-full text-left">
                  <div className="relative">
                    {m.image ? (
                      <img src={m.image} alt={m.name} className="aspect-[16/10] w-full object-cover" loading="lazy" />
                    ) : (
                      <div className="aspect-[16/10] w-full" style={{ background: `${market.accent}1a` }} />
                    )}
                    <span className="absolute left-4 top-4 rounded-full bg-white/95 px-3 py-1.5 text-[11px] font-semibold text-[#1c1a17] shadow-sm">
                      {MEAL_TAG[m.slot - 1] ?? `Sugestão ${m.slot}`}
                    </span>
                    <span
                      className="absolute bottom-4 right-4 rounded-full px-3 py-1.5 text-[12px] font-semibold text-white shadow-sm"
                      style={{ background: market.accent }}
                    >
                      {m.totalCost}
                    </span>
                  </div>
                  <div className="px-5 pb-5 pt-4">
                    <h2 className="font-display text-[22px] font-semibold leading-tight tracking-tight">
                      {m.name}
                    </h2>
                    <p className="mt-2 line-clamp-2 text-sm font-light leading-relaxed text-[#1c1a17]/60">
                      {m.description}
                    </p>
                    <div className="mt-4 flex flex-wrap items-center gap-2 text-[12px] text-[#1c1a17]/60">
                      <span className="whitespace-nowrap rounded-full bg-[#f6f5f2] px-3 py-1.5">{m.time}</span>
                      <span className="whitespace-nowrap rounded-full bg-[#f6f5f2] px-3 py-1.5">{m.servings}</span>
                      <span className="whitespace-nowrap rounded-full bg-[#f6f5f2] px-3 py-1.5">
                        {m.ingredients.length} ingredientes
                      </span>
                    </div>
                  </div>
                </button>
                <div className="flex gap-2.5 px-5 pb-5">
                  <button
                    onClick={() => setOpenMeal(m)}
                    className="flex-1 rounded-full bg-[#1c1a17] py-3.5 text-[13px] font-medium text-white active:scale-[0.98]"
                  >
                    Ver receita
                  </button>
                  <button
                    onClick={() => toggleBasket(m.id)}
                    className={`flex-1 rounded-full py-3.5 text-[13px] font-medium transition-colors active:scale-[0.98] ${
                      inBasket
                        ? "bg-[#e3efe8] text-[#1d5c3f]"
                        : "border border-[#e9e5dd] text-[#1c1a17]/70"
                    }`}
                  >
                    {inBasket ? "✓ Na lista" : "+ Lista"}
                  </button>
                </div>
              </article>
            );
          })}

          {!isLoading && meals.length === 0 && (
            <p className="py-24 text-center text-sm text-[#1c1a17]/45">
              Sem sugestões para este supermercado de momento.
            </p>
          )}

          <p className="pb-2 pt-1 text-center text-[11px] font-light text-[#1c1a17]/40">
            {meals[0] ? `${meals[0].source} · ${validity(meals[0])}` : "Folhetos semanais"} · confirma os preços na loja · imagens ilustrativas
          </p>
        </main>
      </div>

      {/* ─── recipe sheet ─── */}
      {openMeal && (
        <div className="fixed inset-0 z-40 bg-white">
          <div className="sheet-up flex h-full flex-col">
            <div className="relative shrink-0">
              {openMeal.image ? (
                <img src={openMeal.image} alt={openMeal.name} className="h-56 w-full object-cover" />
              ) : (
                <div className="h-56 w-full" style={{ background: `${sheetAccent}1a` }} />
              )}
              <button
                onClick={() => setOpenMeal(null)}
                className="absolute right-4 top-4 flex h-10 w-10 items-center justify-center rounded-full bg-white/95 text-base shadow-md active:scale-95"
                aria-label="Fechar receita"
              >
                ✕
              </button>
              <span
                className="absolute bottom-4 left-4 rounded-full px-3 py-1.5 text-[11px] font-semibold text-white"
                style={{ background: sheetAccent }}
              >
                {openMeal.supermarket}
              </span>
              <span className="absolute bottom-4 right-4 rounded-full bg-[#1c1a17]/85 px-3 py-1.5 text-[12px] font-semibold text-white shadow-sm backdrop-blur-sm">
                {openMeal.totalCost}
              </span>
            </div>

            <div className="flex-1 overflow-y-auto px-5 pb-36 pt-6">
              <h2 className="font-display text-[26px] font-semibold leading-[1.1] tracking-tight">
                {openMeal.name}
              </h2>
              <div className="mt-3 flex flex-wrap items-center gap-2 text-[12px] text-[#1c1a17]/55">
                <span className="whitespace-nowrap rounded-full bg-[#f6f5f2] px-3 py-1.5">{openMeal.time}</span>
                <span className="whitespace-nowrap rounded-full bg-[#f6f5f2] px-3 py-1.5">{openMeal.servings}</span>
              </div>
              <p className="mt-4 text-[15px] font-light leading-relaxed text-[#1c1a17]/65">
                {openMeal.description}
              </p>

              <h3 className="font-display mt-8 text-lg font-semibold tracking-tight">Ingredientes</h3>
              <ul className="mt-3 divide-y divide-[#e9e5dd] border-y border-[#e9e5dd]">
                {openMeal.ingredients.map((ing, j) => (
                  <li key={j} className="py-3.5">
                    <div className="flex items-baseline justify-between gap-3">
                      <span className="text-[15px] text-[#1c1a17]/90">
                        {ing.item}
                        {ing.note ? (
                          <span className="ml-2 rounded-full bg-[#c14e1d]/10 px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wide text-[#c14e1d]">
                            {ing.note}
                          </span>
                        ) : null}
                      </span>
                      <span className="shrink-0 rounded-lg bg-[#f6f5f2] px-2.5 py-1 font-mono-x text-[12px] font-medium">
                        {ing.qty}
                      </span>
                    </div>
                    <p className="mt-1 font-mono-x text-[11px] text-[#1d5c3f]">{ing.price}</p>
                  </li>
                ))}
              </ul>

              <h3 className="font-display mt-9 text-lg font-semibold tracking-tight">
                Preparação
                <span className="ml-2 font-mono-x text-[11px] font-normal text-[#1c1a17]/45">
                  {openMeal.steps.length} passos
                </span>
              </h3>
              <ol className="mt-4 space-y-6">
                {openMeal.steps.map((step, k) => (
                  <li key={k} className="flex gap-4">
                    <span
                      className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-[13px] font-semibold text-white"
                      style={{ background: sheetAccent }}
                    >
                      {k + 1}
                    </span>
                    <p className="text-[15px] font-light leading-relaxed text-[#1c1a17]/85">
                      {step}
                    </p>
                  </li>
                ))}
              </ol>

              <p className="mt-10 border-t border-[#e9e5dd] pt-4 text-[11px] font-light text-[#1c1a17]/40">
                {openMeal.source} · {validity(openMeal)}
                {openMeal.imageCredit && (
                  <a href={openMeal.imageCredit.url} target="_blank" rel="noreferrer" className="mt-1 block underline">
                    Foto: {openMeal.imageCredit.name} / Pexels
                  </a>
                )}
                {openMeal.sources.map((x) => (
                  <a key={x.url} href={x.url} target="_blank" rel="noreferrer" className="mt-1 block underline">
                    {x.title}
                  </a>
                ))}
              </p>
            </div>

            <div className="absolute inset-x-0 bottom-0 border-t border-[#e9e5dd] bg-white/95 px-5 py-4 backdrop-blur">
              <button
                onClick={() => toggleBasket(openMeal.id)}
                className={`w-full rounded-full py-4 text-[14px] font-semibold transition-colors active:scale-[0.98] ${
                  basket.has(openMeal.id)
                    ? "bg-[#e3efe8] text-[#1d5c3f]"
                    : "bg-[#1c1a17] text-white"
                }`}
              >
                {basket.has(openMeal.id)
                  ? "✓ Na lista de compras"
                  : "Adicionar à lista de compras"}
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ─── floating total pill ─── */}
      {basket.size > 0 && !listOpen && !openMeal && (
        <div className="fixed inset-x-0 bottom-0 z-30 px-4 pb-5">
          <button
            onClick={() => setListOpen(true)}
            className="pop-in card-shadow mx-auto flex w-full max-w-md items-center justify-between rounded-full bg-[#1d5c3f] px-6 py-4 text-[13px] font-semibold text-white active:scale-[0.98]"
          >
            <span>
              Lista de compras · {basket.size} {basket.size === 1 ? "refeição" : "refeições"}
            </span>
            <span>≈ {eur(basketTotal)} →</span>
          </button>
        </div>
      )}

      {/* ─── shopping list sheet ─── */}
      {listOpen && (
        <div className="fixed inset-0 z-50 flex flex-col bg-white">
          <div className="sheet-up flex h-full flex-col">
            <div className="flex items-center justify-between border-b border-[#e9e5dd] px-5 py-4">
              <div>
                <h2 className="font-display text-xl font-semibold tracking-tight">Lista de compras</h2>
                <p className="mt-0.5 text-[12px] font-light text-[#1c1a17]/50">
                  {basket.size === 0
                    ? "Ainda está vazia"
                    : `${basket.size} ${basket.size === 1 ? "refeição" : "refeições"} · toca nos itens à medida que compras`}
                </p>
              </div>
              <button
                onClick={() => setListOpen(false)}
                className="flex h-10 w-10 items-center justify-center rounded-full bg-[#f6f5f2] text-base text-[#1c1a17]/70 active:scale-95"
                aria-label="Fechar lista"
              >
                ✕
              </button>
            </div>

            <div className="flex-1 overflow-y-auto px-5 py-4">
              {basket.size === 0 && (
                <p className="py-16 text-center text-sm font-light leading-relaxed text-[#1c1a17]/50">
                  Abre uma receita e toca em<br />“Adicionar à lista de compras”.
                </p>
              )}

              {basketByMarket.map(([marketName, ms]) => {
                const accent =
                  SUPERMARKETS.find((s) => s.name === marketName)?.accent ?? "#1c1a17";
                return (
                  <div key={marketName} className="mt-6 first:mt-0">
                    <div className="flex items-center gap-2">
                      <span className="h-2.5 w-2.5 rounded-full" style={{ background: accent }} />
                      <p className="text-[13px] font-semibold">{marketName}</p>
                    </div>
                    <ul className="mt-2 space-y-1">
                      {ms.flatMap((m) =>
                        m.ingredients.filter((ing) => parseCost(ing.price) !== null).map((ing, j) => {
                          const key = `${m.id}-${j}`;
                          const done = checked.has(key);
                          const cost = parseCost(ing.price);
                          return (
                            <li key={key}>
                              <button
                                onClick={() => toggleChecked(key)}
                                className="flex w-full items-center gap-3 rounded-xl px-2 py-3 text-left active:bg-[#f6f5f2]"
                              >
                                <span
                                  className={`flex h-[22px] w-[22px] shrink-0 items-center justify-center rounded-full border-2 transition-colors ${
                                    done
                                      ? "border-[#1d5c3f] bg-[#1d5c3f] text-white"
                                      : "border-[#1c1a17]/25"
                                  }`}
                                >
                                  {done && "✓"}
                                </span>
                                <span
                                  className={`flex-1 text-[14px] transition-colors ${
                                    done
                                      ? "text-[#1c1a17]/35 line-through"
                                      : "text-[#1c1a17]/85"
                                  }`}
                                >
                                  {ing.item}
                                  <span className="ml-2 font-mono-x text-[11px] text-[#1c1a17]/40">
                                    {ing.qty}
                                  </span>
                                </span>
                                <span
                                  className={`shrink-0 font-mono-x text-[11px] ${
                                    done ? "text-[#1c1a17]/30" : "text-[#1d5c3f]"
                                  }`}
                                >
                                  {cost !== null ? `~${eur(cost)}` : "—"}
                                </span>
                              </button>
                            </li>
                          );
                        })
                      )}
                    </ul>
                  </div>
                );
              })}
            </div>

            {basket.size > 0 && (
              <div className="border-t border-[#e9e5dd] bg-white px-5 py-5">
                <div className="flex items-center justify-between">
                  <span className="text-[13px] font-medium text-[#1c1a17]/60">Total estimado</span>
                  <span className="font-display text-3xl font-semibold tracking-tight">
                    ≈ {eur(basketTotal)}
                  </span>
                </div>
                <p className="mt-1.5 text-[11px] font-light text-[#1c1a17]/45">
                  Ingredientes de despensa não incluídos. Confirma os preços na loja.
                </p>
                <button
                  onClick={() => {
                    setBasket(new Set());
                    setChecked(new Set());
                  }}
                  className="mt-4 w-full rounded-full border border-[#e9e5dd] py-3.5 text-[12px] font-medium text-[#1c1a17]/60 active:scale-[0.98]"
                >
                  Esvaziar lista
                </button>
              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
